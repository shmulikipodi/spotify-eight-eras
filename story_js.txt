  const order = [0,1,2,4,5,6,3,7,8];
  order.forEach(k => {
    const vals = share.map(r => r ? r[k] : null);
    const valid = vals.map((v,i) => [v,i]).filter(z => z[0] != null && D.musich[z[1]] > 2);
    const pk = valid.reduce((a,b) => b[0] > a[0] ? b : a);
    const tot = D.genre.reduce((a,r) => a + r[k], 0);
    const d = document.createElement('div');
    d.innerHTML = `<div class="mh"><b><i style="background:${GC(k)}"></i>${GN[k]}</b><span>${Math.round(tot)} h</span></div><svg></svg><p>Peak ${Math.round(pk[0])}% in ${mLong(M[pk[1]])}</p>`;
    box.appendChild(d);
    const svg = d.querySelector('svg'), W = 340, H = 110, pl = 26, pr = 6, pt = 8, pb = 16, iw = W-pl-pr, ih = H-pt-pb;
    svg.setAttribute('viewBox', `0 0 ${W} ${H}`); svg.style.width = '100%';
    const x = i => pl + i*iw/(NM-1), y = v => pt + ih - Math.min(v, MAXS)/MAXS*ih;
    let s = '';
    D.eras.forEach((e,j) => { if (j % 2) return; const a = M.indexOf(e.a), b = M.indexOf(e.b);
      const xa = a === 0 ? pl : (x(a)+x(a-1))/2, xb = b === NM-1 ? x(NM-1) : (x(b)+x(b+1))/2;
      s += `<rect x="${xa}" y="${pt}" width="${xb-xa}" height="${ih}" fill="var(--rule)" opacity=".45"/>`; });
    [0,30,60].forEach(v => s += `<line x1="${pl}" x2="${pl+iw}" y1="${y(v)}" y2="${y(v)}" stroke="var(--rule)"/><text x="${pl-4}" y="${y(v)+4}" text-anchor="end">${v}</text>`);
    const segs = []; let cur = [];
    vals.forEach((v,i) => { if (v == null) { if (cur.length) segs.push(cur); cur = []; } else cur.push([x(i), y(v)]); }); if (cur.length) segs.push(cur);
    segs.forEach(sg => { s += `<polygon points="${sg[0][0]},${y(0)} ${sg.map(p => p.join(',')).join(' ')} ${sg.at(-1)[0]},${y(0)}" fill="${GC(k)}" opacity=".22"/><polyline points="${sg.map(p => p.join(',')).join(' ')}" fill="none" stroke="${GC(k)}" stroke-width="1.8" stroke-linejoin="round"/>`; });
    s += `<circle cx="${x(pk[1])}" cy="${y(pk[0])}" r="3.5" fill="${GC(k)}" stroke="var(--paper)" stroke-width="1.5"/>`;
    M.forEach((m,i) => { if (m.endsWith('-01')) s += `<text x="${x(i)}" y="${H-3}" text-anchor="middle">${m.slice(2,4)}</text>`; });
    s += `<line class="cx" y1="${pt}" y2="${pt+ih}" stroke="var(--ink)" visibility="hidden"/>`;
    svg.innerHTML = s;
    const cx = svg.querySelector('.cx');
    svg.onmousemove = e => { const p = svgX(svg, e), i = Math.max(0, Math.min(NM-1, Math.round((p.x-pl)/(iw/(NM-1)))));
      cx.setAttribute('x1', x(i)); cx.setAttribute('x2', x(i)); cx.setAttribute('visibility','visible');
      showTip(e, `<b>${GN[k]}</b> · ${mLong(M[i])}<br>${vals[i] == null ? 'Almost nothing played' : `${Math.round(vals[i])}% of the month · ${D.genre[i][k].toFixed(1)} h`}`); };
    svg.onmouseleave = () => { cx.setAttribute('visibility','hidden'); hideTip(); };
  });
  $('#multiP').innerHTML = `<p>The river stacks everything together. Here each genre gets its own line, so you can see when it arrived, when it peaked and whether it came back. The shaded bands are alternating eras.</p><p>Some genres rose once and stayed, like <b>hard rock and metal</b>: rarely above 10% before November 2024, and between 9% and 40% of every month since. Others came and went: <b>electronic</b> was a fifth to a third of your music through June 2024 and almost vanished after early 2025. <b>Hip-hop</b> keeps coming back, in summer 2024, summer 2025 and again in summer 2026. <b>Israeli</b> music comes in short bursts.</p>`;
})();

function transChart(svg, t){
  const W = 1000, pl = 40, pr = 10, pt = 36, sh = 160, gap = 14, hh = 46, pb = 22, H = pt + sh + gap + hh + pb;
  const n = t.weeks.length, iw = W - pl - pr, bw = iw/n;
  svg.setAttribute('viewBox', `0 0 ${W} ${H}`); svg.style.width = '100%'; svg.style.minWidth = '620px';
  const tots = t.wg.map(r => r.reduce((a,b)=>a+b,0)), hmax = Math.max(...tots);
  let s = `<defs><pattern id="h${t.frm}" width="6" height="6" patternUnits="userSpaceOnUse" patternTransform="rotate(45)"><line x1="0" y1="0" x2="0" y2="6" stroke="var(--rule)" stroke-width="2"/></pattern></defs>`;
  [0,50,100].forEach(v => { const yy = pt + sh - v/100*sh; s += `<text x="${pl-6}" y="${yy+4}" text-anchor="end">${v}%</text>`; });
  t.wg.forEach((r,i) => {
    const x0 = pl + i*bw + 1, w = bw - 2, tot = tots[i];
    if (tot < 0.15){ s += `<rect x="${x0}" y="${pt}" width="${w}" height="${sh}" fill="url(#h${t.frm})"/>`; }
    else { let acc = 0; r.forEach((v,k) => { if (v <= 0) return; const h = v/tot*sh; const yy = pt + sh - acc - h; acc += h;
      if (h > 1.2) s += `<rect x="${x0}" y="${yy + 0.6}" width="${w}" height="${h - 1.2}" fill="${GC(k)}" rx="1"/>`; }); }
    const hb = tot/hmax*hh; s += `<rect x="${x0}" y="${pt + sh + gap + hh - hb}" width="${w}" height="${Math.max(hb, tot>0?1:0)}" fill="var(--muted)" rx="1"/>`;
    s += `<rect data-i="${i}" x="${pl + i*bw}" y="${pt-14}" width="${bw}" height="${H - pb - pt + 14}" fill="transparent"/>`;
    const d = t.weeks[i]; if (+d.slice(8) <= 7) s += `<text x="${pl + i*bw + 2}" y="${H-6}">${MON[+d.slice(5,7)-1]}${d.slice(5,7)==='01'?' ’'+d.slice(2,4):''}</text>`;
  });
  s += `<text x="${pl-6}" y="${pt + sh + gap + 10}" text-anchor="end">${Math.round(hmax)}h</text>`;
  const bi = t.weeks.findIndex(w => w >= t.boundary + '-01'), xb = pl + Math.max(0, bi - (t.weeks[bi] > t.boundary + '-01' ? 1 : 0))*bw + (t.weeks[bi] > t.boundary + '-01' ? bw*((7 - (new Date(t.weeks[bi]) - new Date(t.boundary + '-01'))/864e5)/7) : 0);
  s += `<line x1="${xb}" x2="${xb}" y1="${pt-30}" y2="${pt + sh + gap + hh}" stroke="var(--ink)" stroke-width="1.2" stroke-dasharray="3 3"/><text x="${xb + 5}" y="${pt-22}" style="fill:var(--ink);font-family:var(--display);font-style:italic;font-size:13px">era ${t.to+1} →</text>`;
  const marks = {}; t.firsts.forEach(f => { const i = t.weeks.findIndex((w,j) => f[0] >= w && (j === n-1 || f[0] < t.weeks[j+1])); if (i >= 0) (marks[i] ||= []).push(f); });
  Object.entries(marks).forEach(([i]) => { const cx = pl + (+i + .5)*bw; s += `<path d="M${cx-4},${pt-11} L${cx+4},${pt-11} L${cx},${pt-4} Z" fill="var(--ink)"/>`; });
  svg.innerHTML = s;
  svg.onmousemove = e => { const r = e.target.closest('[data-i]'); if (!r) return hideTip(); const i = +r.dataset.i, row = t.wg[i], tot = tots[i];
    const end = new Date(new Date(t.weeks[i]).getTime() + 6*864e5).toISOString().slice(0,10);
    const fs = (marks[i] || []).map(f => `<br>▼ ${esc(f[1])}: first big day (${+f[0].slice(8)} ${MON[+f[0].slice(5,7)-1]})`).join('');
    showTip(e, `<b>Week of ${+t.weeks[i].slice(8)} ${MON[+t.weeks[i].slice(5,7)-1]} ${t.weeks[i].slice(0,4)}</b> · ${tot.toFixed(1)} h` + (tot < 0.15 ? '<br>Almost nothing played' :
      row.map((v,k) => [k,v]).filter(z => z[1] > 0.02).sort((a,b) => b[1]-a[1]).map(([k,v]) => `<div class="r"><span><i style="background:${GC(k)}"></i>${GN[k]}</span><span>${Math.round(100*v/tot)}%</span></div>`).join('')) + fs); };
  svg.onmouseleave = hideTip;
}

        </div></div></div>`;
    }
  });
  box.innerHTML = h;
  box.querySelectorAll('svg[data-t]').forEach(svg => transChart(svg, D.trans[+svg.dataset.t]));
